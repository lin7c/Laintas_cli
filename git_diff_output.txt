+def _diff_refresh_windows(diff: str, total_limit: int = 120) -> list[tuple[int, int]]:

+        new_start = max(1, int(m.group(1)))

+class CacheEntry:
+    __slots__ = ('content', 'mtime', 'size', 'cached_at')
+    content: str

+class FileContentCache:
+    """
+    File read cache for reducing redundant disk I/O and LLM token waste.
+    Uses mtime + size fast validation with TTL expiration.
+    
+    Strategy:
+    1. Fast path: mtime and size unchanged → return cached content
+    2. TTL: auto-expire after 5 minutes to prevent stale cache
+    3. Write-through: any write operation invalidates cache immediately
+    """
+    
+    def __init__(self, ttl_seconds: int = 300):  # 5 minutes default
+        self._cache: dict = {}
+        self._ttl = ttl_seconds
+    
+    def _get_abs_path(self, path: str, cwd: str) -> str:
+        """Normalize path to absolute for cache key."""
+        if os.path.isabs(path):
+            return os.path.normpath(path)
+        return os.path.normpath(os.path.abspath(os.path.join(cwd, path)))
+    
+    def get(self, path: str, cwd: str):
+        """
+        Try to get file content from cache.
+        
+        Returns: (content, was_cached) or None if cache miss
+        """
+        abs_path = self._get_abs_path(path, cwd)
+        
+        if abs_path not in self._cache:
+            return None
+        
+        entry = self._cache[abs_path]
+        
+        # TTL expiration check
+        if time.time() - entry.cached_at > self._ttl:
+            del self._cache[abs_path]
+            return None
+        
+        # Check if file still exists and metadata matches
+        try:
+            stat = os.stat(abs_path)
+        except FileNotFoundError:
+            del self._cache[abs_path]
+            return None
+        
+        # Fast validation: mtime and size both unchanged → content is valid
+        if stat.st_mtime == entry.mtime and stat.st_size == entry.size:
+            return (entry.content, True)
+        
+        # Metadata changed, cache is stale
+        del self._cache[abs_path]
+        return None
+    
+    def put(self, path: str, cwd: str, content: str):
+        """Store file content in cache."""
+        abs_path = self._get_abs_path(path, cwd)
+        try:
+            stat = os.stat(abs_path)
+            self._cache[abs_path] = CacheEntry(
+                content=content,
+                mtime=stat.st_mtime,
+                size=stat.st_size,
+                cached_at=time.time()
+            )
+        except FileNotFoundError:
+            # File was deleted after read? Don't cache
+            pass
+    
+    def invalidate(self, path: str, cwd: str):
+        """Invalidate cache for a file (call after writes)."""
+        abs_path = self._get_abs_path(path, cwd)
+        if abs_path in self._cache:
+            del self._cache[abs_path]
+    
+    def clear(self):
+        """Clear entire cache (for testing)."""
+        self._cache.clear()
